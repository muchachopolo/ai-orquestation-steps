LangSmith Deep Dive

The DevOps platform for LLM applications — debug, test, evaluate, and monitor your AI systems in production.

What is LangSmith?

LangSmith is not a framework for building LLM apps — it's the DevOps platform that sits alongside your application. Think of it as Datadog or New Relic, but purpose-built for AI workflows.

What it does

  • • Traces every LLM call, tool use, and chain step
  • • Evaluates output quality with custom metrics
  • • Manages and versions prompts
  • • Monitors production latency, cost, errors
  • • Collects user feedback for fine-tuning

What it is NOT

  • • Not a framework (use LangChain/LangGraph for that)
  • • Not required to use LangChain
  • • Not only for LangChain — works with any LLM code
  • • Not a model provider
  • • Not open-source (SaaS with self-hosted option)

Any LangChain/LangGraph code traces automatically when env vars are set — zero code changes needed.

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate

# This is automatically traced in LangSmith — no extra code!

    ("system", "You are a helpful assistant."),
    ("human", "{question}")
])

chain = prompt | llm
response = chain.invoke({"question": "What is LangSmith?"})
# ^ This entire execution appears as a trace in your LangSmith dashboard

Manual Tracing with @traceable

For non-LangChain code or custom functions you want to include in traces:

from langsmith import traceable
from openai import OpenAI

client = OpenAI()

@traceable(name="classify_intent", run_type="llm")
def classify_intent(user_message: str) -> str:
    """Classify user intent — traced in LangSmith even without LangChain."""
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": "Classify the intent: greeting, question, complaint, other"},
            {"role": "user", "content": user_message}
        ]
    )
    return response.choices[0].message.content

@traceable(name="handle_user_request", run_type="chain")
def handle_user_request(message: str) -> dict:
    """Parent trace that contains child spans."""
    intent = classify_intent(message)  # child span
    return {"intent": intent, "message": message}

# Running this creates a trace tree in LangSmith:
# handle_user_request (chain)
#   └── classify_intent (llm)
result = handle_user_request("I'm having trouble with my order")

Evaluations

Evaluations let you systematically test your LLM app against datasets. No more "I tried 3 inputs and it looked fine."

Creating Datasets

from langsmith import Client

client = Client()

# Create a dataset programmatically
dataset = client.create_dataset(
    dataset_name="qa-eval-set",
    description="QA pairs for evaluating our support bot"
)

# Add examples to the dataset
examples = [
    {
        "inputs": {"question": "How do I reset my password?"},
        "outputs": {"answer": "Go to Settings > Security > Reset Password"}
    },
    {
        "inputs": {"question": "What are your business hours?"},
        "outputs": {"answer": "Monday to Friday, 9 AM to 6 PM BOT"}
    },
    {
        "inputs": {"question": "Can I get a refund?"},
        "outputs": {"answer": "Yes, within 30 days of purchase with receipt"}
    },
]

for example in examples:
    client.create_example(
        inputs=example["inputs"],
        outputs=example["outputs"],
        dataset_id=dataset.id
    )

Running Evaluations

from langsmith.evaluation import evaluate, LangChainStringEvaluator
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate

# Define the system under test
llm = ChatOpenAI(model="gpt-4o-mini")
prompt = ChatPromptTemplate.from_messages([
    ("system", "You are a customer support agent. Answer concisely."),
    ("human", "{question}")
])
chain = prompt | llm

# Target function that takes dataset inputs and returns a prediction
def predict(inputs: dict) -> dict:
    response = chain.invoke({"question": inputs["question"]})
    return {"answer": response.content}

# Built-in evaluators
evaluators = [
    LangChainStringEvaluator("correctness"),   # Is the answer factually correct?
    LangChainStringEvaluator("helpfulness"),   # Is it helpful to the user?
    LangChainStringEvaluator("relevance"),     # Does it address the question?
]

# Run the evaluation
results = evaluate(
    predict,
    data="qa-eval-set",
    evaluators=evaluators,
    experiment_prefix="support-bot-v2",
    metadata={"model": "gpt-4o-mini", "version": "2.0"}
)

Custom Evaluators

from langsmith.schemas import Example, Run

def conciseness_evaluator(run: Run, example: Example) -> dict:
    """Custom evaluator: penalize answers that are too long."""
    prediction = run.outputs.get("answer", "")
    word_count = len(prediction.split())

    if word_count <= 30:
        score = 1.0
    elif word_count <= 60:
        score = 0.7
    else:
        score = 0.3

    return {
        "key": "conciseness",
        "score": score,
        "comment": f"Answer has {word_count} words"
    }

def contains_no_hallucination(run: Run, example: Example) -> dict:
    """Check that the answer doesn't contradict the reference."""
    prediction = run.outputs.get("answer", "")
    reference = example.outputs.get("answer", "")

    # Simple keyword overlap check (in production, use LLM-as-judge)
    ref_keywords = set(reference.lower().split())
    pred_keywords = set(prediction.lower().split())
    overlap = len(ref_keywords & pred_keywords) / max(len(ref_keywords), 1)

    return {
        "key": "no_hallucination",
        "score": 1.0 if overlap > 0.3 else 0.0,
        "comment": f"Keyword overlap: {overlap:.2f}"
    }

# Use custom evaluators in evaluation
results = evaluate(
    predict,
    data="qa-eval-set",
    evaluators=[conciseness_evaluator, contains_no_hallucination],
    experiment_prefix="support-bot-v2-custom-eval"
)

Prompt Management

Version your prompts like code. Deploy, rollback, and A/B test different versions without redeploying your app.

Push & Pull Prompts

from langchain import hub
from langchain_core.prompts import ChatPromptTemplate

# Create and push a prompt to LangSmith Hub
support_prompt = ChatPromptTemplate.from_messages([
    ("system", """You are a customer support agent for Caseritas Store.
Rules:
- Always respond in Spanish (Bolivia)
- Be concise (max 2 sentences)
- If you don't know, say "No tengo esa información, te conecto con un agente"
- Never make up policies"""),
    ("human", "{question}")
])

# Push to your organization's prompt repo
hub.push("my-org/support-agent-v1", support_prompt, new_repo_is_public=False)

# Pull in production code — always gets latest committed version
production_prompt = hub.pull("my-org/support-agent-v1")

# Pull a specific commit (for pinned deployments)
pinned_prompt = hub.pull("my-org/support-agent-v1:abc123def")

A/B Testing Prompts

import random
from langchain import hub
from langchain_openai import ChatOpenAI
from langsmith import traceable

llm = ChatOpenAI(model="gpt-4o-mini")

# Load two prompt versions
prompt_a = hub.pull("my-org/support-agent-v1")  # Original
prompt_b = hub.pull("my-org/support-agent-v2")  # New version with examples

@traceable(name="support_ab_test")
def answer_with_ab_test(question: str) -> dict:
    """Route 50/50 between prompt versions, tracked in LangSmith."""
    variant = "A" if random.random() < 0.5 else "B"
    prompt = prompt_a if variant == "A" else prompt_b

    chain = prompt | llm
    response = chain.invoke({"question": question})

    return {
        "answer": response.content,
        "variant": variant  # This metadata appears in LangSmith traces
    }

from langsmith import Client
from langsmith.run_helpers import get_current_run_tree

client = Client()

@traceable(name="chat_response")
def get_chat_response(user_message: str) -> dict:
    """Generate response and return run_id for feedback."""
    # Get the current run's ID so we can attach feedback later
    run_tree = get_current_run_tree()
    run_id = str(run_tree.id) if run_tree else None

    # ... your LLM logic here ...
    response = "Tu pedido llegará en 15 minutos"

    return {"response": response, "run_id": run_id}

# In your API endpoint, after user gives thumbs up/down:
def submit_feedback(run_id: str, score: int, comment: str = ""):
    """Called when user clicks thumbs up (1) or thumbs down (0)."""
    client.create_feedback(
        run_id=run_id,
        key="user_rating",
        score=score,           # 0 or 1 for thumbs, or 1-5 for stars
        comment=comment,       # Optional user comment
    )

# Example: user liked the response

    run_id="xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx",
    score=1,
    comment="Rápido y preciso"
)

"""
Multi-Agent Content Creation Pipeline with LangGraph.
Requirements: pip install langgraph langchain-openai langsmith tavily-python
Env vars: OPENAI_API_KEY, TAVILY_API_KEY, LANGCHAIN_TRACING_V2=true,
          LANGCHAIN_API_KEY, LANGCHAIN_PROJECT=content-pipeline
"""
import operator
from typing import Annotated, TypedDict
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from langsmith import traceable
from tavily import TavilyClient

# ─── State Definition ───────────────────────────────────────────────────────

class PipelineState(TypedDict):
    topic: str
    research: str
    draft: str
    feedback: str
    revision_count: Annotated[int, operator.add]
    final_article: str
    status: str

# ─── LLM Setup ──────────────────────────────────────────────────────────────

llm = ChatOpenAI(model="gpt-4o", temperature=0.7)
tavily = TavilyClient()

# ─── Agent Nodes ─────────────────────────────────────────────────────────────

@traceable(name="research_agent", run_type="chain")
def research_agent(state: PipelineState) -> dict:
    """Search for information on the topic and compile research notes."""
    topic = state["topic"]

    # Use Tavily to search the web
    search_results = tavily.search(
        query=topic,
        max_results=5,
        include_raw_content=False
    )

    # Compile research from search results
    sources = []
    for result in search_results.get("results", []):
        sources.append(f"- {result['title']}: {result['content'][:200]}")

    research_summary = llm.invoke([
        SystemMessage(content="You are a research analyst. Summarize the following search results into clear, factual research notes. Include key statistics and quotes."),
        HumanMessage(content=f"Topic: {topic}\n\nSearch Results:\n" + "\n".join(sources))
    ])

    return {"research": research_summary.content, "status": "researched"}


@traceable(name="writer_agent", run_type="chain")
def writer_agent(state: PipelineState) -> dict:
    """Write an article draft based on the research."""
    feedback_context = ""
    if state.get("feedback"):
        feedback_context = f"\n\nPREVIOUS FEEDBACK TO ADDRESS:\n{state['feedback']}\n\nPREVIOUS DRAFT:\n{state['draft']}"

    response = llm.invoke([
        SystemMessage(content="""You are an expert content writer. Write a compelling, well-structured article.
Rules:
- Use clear headers (##) for sections
- Include an engaging introduction and conclusion
- Cite research findings naturally in the text
- Keep paragraphs short (3-4 sentences max)
- Total length: 500-800 words"""),
        HumanMessage(content=f"Topic: {state['topic']}\n\nResearch Notes:\n{state['research']}{feedback_context}")
    ])

    return {"draft": response.content, "revision_count": 1, "status": "drafted"}


@traceable(name="editor_agent", run_type="chain")
def editor_agent(state: PipelineState) -> dict:
    """Review the draft and provide feedback or approve."""
    response = llm.invoke([
        SystemMessage(content="""You are a senior editor. Review this article draft and decide:
1. If quality is HIGH (clear, engaging, well-structured, factual) → respond ONLY with: APPROVED
2. If quality needs improvement → provide specific, actionable feedback (max 3 bullet points)

Be strict but fair. Articles must be publication-ready."""),
        HumanMessage(content=f"Topic: {state['topic']}\n\nDraft:\n{state['draft']}")
    ])

    content = response.content.strip()

    if "APPROVED" in content.upper() and len(content) < 50:
        return {"feedback": "", "status": "approved"}
    else:
        return {"feedback": content, "status": "needs_revision"}


@traceable(name="publisher_agent", run_type="chain")
def publisher_agent(state: PipelineState) -> dict:
    """Format the approved article for publication."""
    response = llm.invoke([
        SystemMessage(content="""You are a publishing formatter. Take this approved article and:
1. Add a compelling title (# Title)
2. Add a 1-sentence meta description
3. Add 3-5 relevant tags
4. Format with proper markdown
5. Add a "Key Takeaways" section at the end (3 bullets)

Output the final formatted article ready for publication."""),
        HumanMessage(content=state["draft"])
    ])

    return {"final_article": response.content, "status": "published"}

# ─── Routing Logic ───────────────────────────────────────────────────────────

def should_continue_editing(state: PipelineState) -> str:
    """Route based on editor decision — max 3 revisions."""
    if state["status"] == "approved":
        return "publish"
    elif state.get("revision_count", 0) >= 3:
        return "publish"  # Force publish after 3 revisions
    else:
        return "revise"

# ─── Graph Construction ──────────────────────────────────────────────────────

workflow = StateGraph(PipelineState)

# Add nodes
workflow.add_node("research", research_agent)
workflow.add_node("write", writer_agent)
workflow.add_node("edit", editor_agent)
workflow.add_node("publish", publisher_agent)

# Add edges
workflow.set_entry_point("research")
workflow.add_edge("research", "write")
workflow.add_edge("write", "edit")

# Conditional edge: editor can loop back to writer or proceed to publish
workflow.add_conditional_edges(
    "edit",
    should_continue_editing,
    {"revise": "write", "publish": "publish"}
)

workflow.add_edge("publish", END)

# Compile with checkpointing (enables human-in-the-loop)
memory = MemorySaver()
app = workflow.compile(
    checkpointer=memory,
    interrupt_before=["publish"]  # Human approval gate before publishing
)

# ─── Run the Pipeline ────────────────────────────────────────────────────────

def run_pipeline(topic: str) -> str:
    """Execute the full content pipeline with human approval."""
    config = {"configurable": {"thread_id": "content-001"}}

    # Start the pipeline — will pause before "publish"
    result = app.invoke(
        {"topic": topic, "revision_count": 0},
        config=config
    )

    print(f"\n{'='*60}")
    print(f"Pipeline paused for human approval.")
    print(f"Status: {result['status']}")
    print(f"Revisions: {result['revision_count']}")
    print(f"{'='*60}")
    print(f"\nDraft preview:\n{result['draft'][:500]}...")

    # Simulate human approval (in production, this waits for API call)
    approval = input("\n\nApprove for publication? (yes/no): ")

    if approval.lower() == "yes":
        # Resume the pipeline
        final = app.invoke(None, config=config)
        print(f"\n\n{'='*60}")
        print("PUBLISHED!")
        print(f"{'='*60}")
        print(final["final_article"])
        return final["final_article"]
    else:
        print("Publication cancelled.")
        return ""

if __name__ == "__main__":
    article = run_pipeline("The impact of AI agents on software development in 2025")

│         └─ format_article          llm       2.1s    1600 tokens        │
│                                                                         │
│  Total tokens: 8,430  │  Total cost: $0.18  │  Revisions: 2           │

What to look for in production: Monitor average revision_count — if it creeps above 2.5, your writer prompt needs tuning. Track editor approval rate — below 40% on first pass means research quality is dropping. Set alerts for total pipeline cost above $0.50 (usually means infinite loops).

Best of Both Worlds

CLI + Programmatic Agents

Know when to use Claude Code (CLI) vs LangGraph (programmatic) — and how to combine them for maximum impact.

CLI Orchestration (Claude Code)

Best for development-time tasks where a human is in the loop.

  • ✓ Building entire projects from specs
  • ✓ Prototyping and exploring ideas
  • ✓ Writing documentation
  • ✓ Debugging complex issues
  • ✓ One-shot code generation
  • ✓ Refactoring and code reviews
  • ✓ File system operations
  • ✓ Multi-tool orchestration (git, npm, docker)

Programmatic (LangGraph)

Best for runtime systems that operate autonomously.

  • ✓ Production API endpoints
  • ✓ User-facing chatbots and agents
  • ✓ Persistent multi-turn workflows
  • ✓ Background processing pipelines
  • ✓ Scheduled automation
  • ✓ Multi-agent collaboration systems
  • ✓ Human-in-the-loop approval flows
  • ✓ Scalable concurrent execution

Criteria CLI (Claude Code) LangGraph

The Hybrid Pattern

The most powerful approach: use CLI to BUILD the system, then LangGraph runs it autonomously, and LangSmith monitors everything.

Phase 1: Scaffold with CLI

Claude Code builds the project structure, writes the LangGraph agents, creates tests.

claude "Build a LangGraph agent that handles customer support for an
e-commerce platform. It should classify intent, search FAQs, handle
returns, and escalate to humans. Include full test suite."

# Claude Code creates:
# - src/agents/support_graph.py (LangGraph implementation)
# - src/agents/nodes/ (individual agent nodes)
# - src/tools/ (FAQ search, order lookup tools)
# - tests/ (unit + integration tests)
# - docker-compose.yml (for local dev)
# - .env.example

Phase 2: LangGraph Runs in Production

The built system deploys and handles real user requests autonomously at scale.

# FastAPI endpoint powered by LangGraph
from fastapi import FastAPI
from src.agents.support_graph import support_agent

app = FastAPI()

@app.post("/api/support")
async def handle_support(message: str, thread_id: str):
    config = {"configurable": {"thread_id": thread_id}}
    result = await support_agent.ainvoke(
        {"messages": [HumanMessage(content=message)]},
        config=config
    )
    return {"response": result["messages"][-1].content}

Phase 3: LangSmith Monitors

Every production request is traced. Dashboards show health. Alerts fire on anomalies.

# Automatic monitoring — just set env vars
LANGCHAIN_TRACING_V2=true
LANGCHAIN_PROJECT=support-agent-prod

# LangSmith shows:
# - 2,340 traces today
# - Avg latency: 2.1s
# - Error rate: 0.3%
# - User satisfaction: 4.1/5
# - Escalation rate: 12% (↓ from 18% last week)

Phase 4: CLI Iterates Based on Data

Use Claude Code again to improve the system based on LangSmith insights.

claude "LangSmith shows our FAQ retrieval has 0.6 relevance score —
users are asking questions our FAQ doesn't cover. Add a fallback that
searches our knowledge base with semantic search. The top failing
queries are: [paste from LangSmith]"

# Claude Code updates the agent → redeploy → monitor again
# Continuous improvement loop!

TL;DR: CLI is your builder, LangGraph is your runner, LangSmith is your observer. Together, they form a complete AI engineering workflow.

LangGraph Patterns

langchain-ai.github.io/langgraph 
from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver

# 1. Define State
class AgentState(TypedDict):
    messages: Annotated[list, operator.add]  # append-only
    current_step: str
    result: str

# 2. Define Nodes (functions that take state, return partial state)
def step_one(state: AgentState) -> dict:
    return {"current_step": "one", "result": "did step one"}

def step_two(state: AgentState) -> dict:
    return {"current_step": "two", "result": "did step two"}

# 3. Define Routing
def router(state: AgentState) -> str:
    if state["result"] == "needs_retry":
        return "step_one"
    return "step_two"

# 4. Build Graph
graph = StateGraph(AgentState)
graph.add_node("step_one", step_one)
graph.add_node("step_two", step_two)

graph.set_entry_point("step_one")
graph.add_conditional_edges("step_one", router, {"step_one": "step_one", "step_two": "step_two"})
graph.add_edge("step_two", END)

# 5. Compile
memory = MemorySaver()
app = graph.compile(checkpointer=memory)

# 6. Run
result = app.invoke(
    {"messages": [], "current_step": "", "result": ""},
    config={"configurable": {"thread_id": "123"}}
)

final = app.invoke(None, config=config)

app.update_state(config, {"input": "just list the records instead"})
final = app.invoke(None, config=config)

from langsmith import Client
from langsmith.evaluation import evaluate
from langsmith.schemas import Example, Run

client = Client()

# 1. Create dataset
dataset = client.create_dataset("my-eval-set")
client.create_example(
    inputs={"question": "What is 2+2?"},
    outputs={"answer": "4"},
    dataset_id=dataset.id
)

# 2. Define target function
def predict(inputs: dict) -> dict:
    # Your LLM chain here
    response = chain.invoke(inputs)
    return {"answer": response}

# 3. Define evaluator
def correctness(run: Run, example: Example) -> dict:
    predicted = run.outputs.get("answer", "")
    expected = example.outputs.get("answer", "")
    return {
        "key": "correct",
        "score": 1.0 if expected.lower() in predicted.lower() else 0.0
    }

# 4. Run evaluation
results = evaluate(
    predict,
    data="my-eval-set",
    evaluators=[correctness],
    experiment_prefix="v2-test"
)
print(f"Score: {results.aggregate_metrics}")
